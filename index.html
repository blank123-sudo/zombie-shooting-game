const WebSocket = require('ws'); const http = require('http'); const server = http.createServer((req, res) => { res.writeHead(200, { 'Content-Type': 'text/plain' }); res.end('Zombie Survival Multiplayer Room Server is Running\n'); }); const wss = new WebSocket.Server({ server }); // Rooms map: roomId -> { players: { id: { ws, x, y, angle, hp, wave, name } }, matchStarted: false, hostId: null } const rooms = {}; wss.on('connection', (ws) => { let currentRoomId = null; let playerId = 'player_' + Math.random().toString(36.substring(2, 9)); ws.on('message', (message) => { try { let data = JSON.parse(message); if (data.type === 'CREATE_ROOM' || data.type === 'JOIN_ROOM') { let roomId = (data.roomId || '').trim().toUpperCase(); if (!roomId) return; // Leave previous room if any if (currentRoomId && rooms[currentRoomId]) { delete rooms[currentRoomId].players[playerId]; if (Object.keys(rooms[currentRoomId].players).length === 0) { delete rooms[currentRoomId]; } else if (rooms[currentRoomId].hostId === playerId) { rooms[currentRoomId].hostId = Object.keys(rooms[currentRoomId].players)[0]; } broadcastRoomState(currentRoomId); } currentRoomId = roomId; if (!rooms[currentRoomId]) { rooms[currentRoomId] = { players: {}, matchStarted: false, hostId: playerId }; } rooms[currentRoomId].players[playerId] = { id: playerId, name: data.name || 'Survivor', x: 8, y: 12, angle: 0, hp: 100, wave: 1 }; // If host left or room was empty, ensure host is set correctly if (!rooms[currentRoomId].hostId) { rooms[currentRoomId].hostId = playerId; } broadcastRoomState(currentRoomId); } if (data.type === 'MOVE' && currentRoomId && rooms[currentRoomId]) { let p = rooms[currentRoomId].players[playerId]; if (p) { p.x = data.x; p.y = data.y; p.angle = data.angle; p.hp = data.hp; p.wave = data.wave; broadcastRoomState(currentRoomId); } } if (data.type === 'START_MATCH' && currentRoomId && rooms[currentRoomId]) { // Only allow the designated host to start the match if (rooms[currentRoomId].hostId === playerId) { rooms[currentRoomId].matchStarted = true; broadcastRoomState(currentRoomId); } } } catch (e) { console.error("Error parsing message:", e); } }); ws.on('close', () => { if (currentRoomId && rooms[currentRoomId]) { delete rooms[currentRoomId].players[playerId]; if (Object.keys(rooms[currentRoomId].players).length === 0) { delete rooms[currentRoomId]; } else if (rooms[currentRoomId].hostId === playerId) { rooms[currentRoomId].hostId = Object.keys(rooms[currentRoomId].players)[0]; } broadcastRoomState(currentRoomId); } }); }); function broadcastRoomState(roomId) { if (!rooms[roomId]) return; let roomData = rooms[roomId]; let payload = JSON.stringify({ type: 'STATE', matchStarted: roomData.matchStarted, hostId: roomData.hostId, players: roomData.players }); wss.clients.forEach(client => { if (client.readyState === WebSocket.OPEN) { // Find if this client belongs to this room // A simple way is broadcasting to clients tracking this room context client.send(payload); } }); } const PORT = process.env.PORT || 10000; server.listen(PORT, () => { console.log(`Server listening on port ${PORT}`); });
